          if(typeof parsed.score === 'number'){
            results.push({name: (parsed.name||'名無し工員').slice(0,10), score: parsed.score, isMe: key===myKey});
          }
        }catch(err){ /* skip malformed entry */ }
      }
    }
  }catch(e){ /* leaderboard fetch failed */ }
  results.sort((a,b)=>b.score-a.score);
  return results.slice(0,100);
}

/* weighted pick of 10 unique words, favoring past mistakes */
function buildQueue(){
  const pool = QUESTION_BANK.slice();
  const picks = [];
  const n = Math.min(10,pool.length);
  for(let i=0;i<n;i++){
    let weights = pool.map(w=>1+mistakeWeight[w[0]]*3);
    let total = weights.reduce((a,b)=>a+b,0);
    let r = Math.random()*total;
    let idx=0;
    for(;idx<weights.length;idx++){ r-=weights[idx]; if(r<=0) break; }
    if(idx>=pool.length) idx=pool.length-1;
    picks.push(pool[idx]);
    pool.splice(idx,1);
  }
  return picks;
}

function nextRound(){
  if(state.round >= state.totalRounds){ endGame(); return; }
  state.locked = false;
  const [word, jp] = state.queue[state.round];
  state.currentWord = word;
  state.tokens = getTokens(word);
  state.isPhraseRound = isPhrase(word);
  state.currentIdx = 0;
  state.roundMistakeCount = 0;
  el('order-jp').textContent = jp;
  updateHud();
  buildSlots(word);
  buildBelt(word);
}

function buildSlots(word){
  const box = el('slots');
  box.innerHTML='';

  const tokens = state.tokens;
  const n = tokens.length;
  const GAP = 6;
  const availW = box.clientWidth - 12; // minus horizontal padding
  const BASE_H = 88, BASE_FONT = 40;

  if(state.isPhraseRound){
    // each slot's natural width depends on its word's length (wide enough to show the word once filled)
    const CHAR_W = BASE_FONT*0.62;
    const PAD = 22;
    const naturalWidths = tokens.map(t => Math.max(BASE_H*0.7, t.length*CHAR_W + PAD));
    const neededW = naturalWidths.reduce((a,b)=>a+b,0) + (n-1)*GAP;
    const scale = Math.min(1, availW / Math.max(neededW,1));
    const fontSize = Math.max(11, Math.round(BASE_FONT*scale));
    const slotH = Math.max(24, Math.round(BASE_H*scale));

    tokens.forEach((t,i)=>{
      const d=document.createElement('div');
      d.className='slot phrase-slot';
      d.id='slot-'+i;
      d.style.width = Math.round(naturalWidths[i]*scale)+'px';
      d.style.height = slotH+'px';
      d.style.fontSize = fontSize+'px';
      box.appendChild(d);
    });
  }else{
    const BASE_W = 76;
    const neededW = n*BASE_W + (n-1)*GAP;
    const scale = Math.min(1, availW / Math.max(neededW,1));
    const slotW = Math.max(22, Math.round(BASE_W*scale));
    const slotH = Math.max(26, Math.round(BASE_H*scale));
    const fontSize = Math.max(12, Math.round(BASE_FONT*scale));

    for(let i=0;i<n;i++){
      const d=document.createElement('div');
      d.className='slot';
      d.id='slot-'+i;
      d.style.width = slotW+'px';
      d.style.height = slotH+'px';
      d.style.fontSize = fontSize+'px';
      box.appendChild(d);
    }
  }
}

const IN_MS = 650;
const OUT_MS = 550;
function calcHoldMs(word){
  const tokenCount = getTokens(word).length;
  const base = isPhrase(word) ? (4500 + tokenCount*2200) : (4000 + word.length*1000);
  const reduced = base - (state.level-1)*250 - wallet.evolutionStage*450;
  return Math.max(5000, Math.min(15000, reduced));
}

function tileNaturalSize(token){
  if(state.isPhraseRound){
    const font = 26;
    const w = Math.max(52, token.length*font*0.62 + 30);
    return {w, h:58, font};
  }
  return {w:64, h:73, font:31};
}

function buildBelt(word){
  const track = el('belt-wrap');
  // remove old tiles
  state.tiles.forEach(t=>t.el && t.el.remove());
  state.tiles = [];
  clearRoundTimers();

  let tokens = state.tokens.slice();
  // shuffle
  for(let i=tokens.length-1;i>0;i--){ const j=Math.floor(Math.random()*(i+1)); [tokens[i],tokens[j]]=[tokens[j],tokens[i]]; }

  const beltW = track.clientWidth;
  const beltH = track.clientHeight;
  const count = tokens.length;

  const TILE_H = tileNaturalSize(tokens[0]).h;
  const BASE_FONT = tileNaturalSize(tokens[0]).font;
  const PAD = 12;
  const GAP_UNSCALED = state.isPhraseRound ? 10 : 12;
  const availW = beltW - 24;
  const availH = beltH - 16;

  // try 1, 2, then 3 rows; pick the fewest rows whose resulting scale stays reasonably readable
  let best = null;
  for(let rows=1; rows<=3; rows++){
    const rowItems = Array.from({length:rows}, ()=>[]);
    tokens.forEach((t,i)=>{ rowItems[i%rows].push(t); });
    let maxRowNaturalW = 0;
    rowItems.forEach(items=>{
      const w = items.reduce((sum,t)=>sum+tileNaturalSize(t).w,0) + GAP_UNSCALED*Math.max(0,items.length-1);
      if(w>maxRowNaturalW) maxRowNaturalW = w;
    });
    const scaleW = Math.min(1, availW / Math.max(maxRowNaturalW,1));
    const scaleH = Math.min(1, (availH/rows - PAD) / TILE_H);
    const scale = Math.max(0.32, Math.min(scaleW, scaleH));
    best = {rows, scale, rowItems};
    if(scale >= 0.62 || rows===3) break; // good enough readability, or out of row budget
  }
  const {rows, scale, rowItems} = best;

  const ROW_PAD = 10; // tight vertical gap between rows, independent of belt height
  const rowHeight = Math.min(availH/rows, TILE_H*scale + ROW_PAD);
  const totalRowsHeight = rowHeight*rows;
  const startY = Math.max(rowHeight/2, (beltH-totalRowsHeight)/2 + rowHeight/2);

  let globalIdx = 0;
  rowItems.forEach((items, r)=>{
    const itemWidths = items.map(t=>tileNaturalSize(t).w*scale);
    const rowWidth = itemWidths.reduce((a,b)=>a+b,0) + GAP_UNSCALED*scale*Math.max(0,items.length-1);
    let cursorX = Math.max(14, (beltW-rowWidth)/2);
    const laneY = startY + r*rowHeight;
    items.forEach((token, idxInRow)=>{
      const natSize = tileNaturalSize(token);
      const scaledW = natSize.w*scale;
      const d = document.createElement('div');
      d.className = 'tile' + (state.isPhraseRound ? ' word-tile' : '');
      d.textContent = token;
      d.style.transition='none';
      d.style.width = natSize.w+'px';
      d.style.height = natSize.h+'px';
      d.style.fontSize = natSize.font+'px';
      d.style.transform = `translateY(-50%) scale(${scale})`;
      d.style.left = (beltW+120+globalIdx*24)+'px';
      d.style.top = laneY+'px';
      d.style.pointerEvents='none';
      track.appendChild(d);
      const targetX = cursorX + scaledW/2 - natSize.w/2; // account for CSS transform-origin center scaling
      const tileObj = {el:d, token:token, alive:true, targetX, laneY, scale};
      d.addEventListener('click', ()=>handleTileClick(tileObj));
      d.addEventListener('touchstart', (e)=>{e.preventDefault(); handleTileClick(tileObj);},{passive:false});
      state.tiles.push(tileObj);
      cursorX += scaledW + GAP_UNSCALED*scale;
      globalIdx++;
    });
  });

  runBeltIn(word);
}

function speakWord(word){
  try{
    if(!('speechSynthesis' in window)) return;
    window.speechSynthesis.cancel();
    const utter = new SpeechSynthesisUtterance(word);
    utter.lang = 'en-US';
    utter.rate = 0.85;
    utter.pitch = 1.05;
    window.speechSynthesis.speak(utter);
  }catch(e){ /* speech synthesis unavailable, ignore */ }
}

  el('gauge-money').classList.remove('money-hidden');

function runBeltIn(word){
  state.phase = 'in';
  hideHoldBar();
  showMoneyGauge();
  setBeltMoving(true);
  requestAnimationFrame(()=>{
    state.tiles.forEach(t=>{
      t.el.style.transition = `left ${IN_MS}ms cubic-bezier(.2,.7,.3,1), top ${IN_MS}ms ease`;
      t.el.style.left = t.targetX+'px';
      t.el.style.top = t.laneY+'px';
    });
  });
  state.phaseTimeoutId = setTimeout(()=>{
    state.tiles.forEach(t=>{ if(t.alive) t.el.style.pointerEvents='auto'; });
    speakWord(word);
    startHold(word);
  }, IN_MS+40);
}

function startHold(word){
  state.phase = 'hold';
  setBeltMoving(false);
  hideMoneyGauge();
  const holdMs = calcHoldMs(word);
  showHoldBar(holdMs);
  state.holdTimeoutId = setTimeout(()=>{
    if(state.phase==='hold'){ runBeltOut(); }
  }, holdMs);
}

function runBeltOut(){
  state.phase = 'out';
  hideHoldBar();
  showMoneyGauge();
  setBeltMoving(true);
  state.tiles.forEach(t=>{
    if(!t.alive) return;
    t.el.style.pointerEvents='none';
    t.el.style.transition = `left ${OUT_MS}ms cubic-bezier(.5,0,.9,.4), top ${OUT_MS}ms ease`;
    t.el.style.left = '-70px';
  });
  state.phaseTimeoutId = setTimeout(()=>{ missRound(); }, OUT_MS+80);
}

function handleTileClick(tileObj){
  if(state.locked || !tileObj.alive || state.phase!=='hold') return;
  const need = state.tokens[state.currentIdx];
  ac(); // resume/create audio ctx on user gesture
  if(tileObj.token === need){
    tileObj.alive = false;
    state.totalLetters++;
    placeIntoSlot(tileObj, state.currentIdx);
    sfxCorrectLetter(state.currentIdx);
    state.currentIdx++;
    if(state.currentIdx >= state.tokens.length){
      state.locked = true;
      clearRoundTimers();
      hideHoldBar();
      setTimeout(completeWord, 380);
    }
  }else{
    state.totalMistakes++;
    state.combo = 0;
    state.roundMistakeCount++;
    mistakeWeight[state.currentWord] = (mistakeWeight[state.currentWord]||0) + 1;
    updateHud();
    tileObj.el.classList.add('wrong');
    sfxWrong();
    shakeScreen();
    setTimeout(()=>tileObj.el.classList.remove('wrong'), 350);
  }
}


function placeIntoSlot(tileObj, idx){
  const slot = el('slot-'+idx);
  const slotRect = slot.getBoundingClientRect();
  const wrapRect = el('belt-wrap').getBoundingClientRect();
  tileObj.el.style.transition = '';
  tileObj.el.classList.add('flyaway');
  const unscaledW = tileObj.el.offsetWidth;
  const targetCenterX = slotRect.left - wrapRect.left + slot.offsetWidth/2;
  tileObj.el.style.left = (targetCenterX - unscaledW/2)+'px';
  tileObj.el.style.top = (slotRect.top - wrapRect.top + slot.offsetHeight/2)+'px';
  setTimeout(()=>{
    slot.textContent = tileObj.token;
    slot.classList.add('filled','pop');
    tileObj.el.remove();
  }, 340);
}

function wordDifficulty(word){
  const tokenCount = getTokens(word).length;
  const base = isPhrase(word) ? tokenCount*2.2 : word.length;
  return base + (state.level-1)*1.5 + wallet.evolutionStage*1.5;
}

function completeWord(){
  showMoneyGauge();
  const comboBefore = state.combo;
  state.combo++;
  state.maxCombo = Math.max(state.maxCombo, state.combo);

  const D = wordDifficulty(state.currentWord);
  const diffMult = 1 + D*0.05;
  const accuracyMult = state.roundMistakeCount===0 ? 1.25 : (state.roundMistakeCount<=2 ? 1.0 : 0.75);
  const comboMult = 1 + Math.min(comboBefore,10)*0.06;
  const evolutionMoneyMult = 1 + wallet.evolutionStage*0.2; // factory evolution boosts earnings too
  const earned = Math.round(90 * diffMult * accuracyMult * comboMult * evolutionMoneyMult);

  state.round++;

  sfxWordComplete();
  popStamp();
  burstParticles();
  addMoney(earned);
  updateHud();

  // level up every 3 completed rounds
  if(state.round % 3 === 0 && state.round < state.totalRounds){
    state.level++;
    setTimeout(()=>{ sfxLevelUp(); showLevelUp(); }, 500);
  }

  setTimeout(nextRound, state.round % 3 === 0 && state.round < state.totalRounds ? 1400 : 700);
}

function missRound(){
  state.combo = 0;
  state.totalMistakes++;
  state.round++;
  mistakeWeight[state.currentWord] = (mistakeWeight[state.currentWord]||0) + 1;

  const D = wordDifficulty(state.currentWord);
  const lost = -Math.round(90 * (1+D*0.03) * 0.4);

  sfxWrong();
  popMissStamp();
  shakeScreen();
  addMoney(lost);
  updateHud();

  if(state.round % 3 === 0 && state.round < state.totalRounds){
    state.level++;
    setTimeout(()=>{ sfxLevelUp(); showLevelUp(); }, 500);
  }

  setTimeout(nextRound, state.round % 3 === 0 && state.round < state.totalRounds ? 1400 : 700);
}

function endGame(){
  clearRoundTimers();
  hideHoldBar();
  sfxGameOver();
  const acc = state.totalLetters+state.totalMistakes>0 ? Math.round(100*state.totalLetters/(state.totalLetters+state.totalMistakes)) : 100;

  const thisGameEarnings = Math.max(0, Math.round(state.gameEarnings||0));
  wallet.bestGameEarnings = Math.max(wallet.bestGameEarnings, thisGameEarnings);
  savePersistentWallet();

  const evolved = evolveAfterGame();
  submitScoreToLeaderboard();

  el('result-money').textContent = '¥'+wallet.money;
  el('result-earned').textContent = '¥'+thisGameEarnings;
  el('result-level').textContent = state.level;
  el('result-accuracy').textContent = acc+'%';
  el('result-maxcombo').textContent = state.maxCombo;
  el('result-games').textContent = wallet.gamesPlayed;

  let rank = '見習い工員';
  if(wallet.bestScore>=1200) rank='伝説の職人マイスター';
  else if(wallet.bestScore>=700) rank='工場長';
  else if(wallet.bestScore>=400) rank='熟練職人';
  else if(wallet.bestScore>=200) rank='一人前の工員';
  el('result-rank').textContent = rank;

  const banner = el('result-evolve-banner');
  if(evolved){
    banner.textContent = '🏭 工場が進化した！(Lv.'+wallet.evolutionStage+')';
    banner.classList.remove('hidden');
    setTimeout(sfxEvolve, 400);
  }else{
    banner.classList.add('hidden');
  }

  showScreen('result');
}
